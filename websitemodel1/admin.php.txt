function load_settings() {
    $s = default_settings();
    if (file_exists(SETTINGS_FILE)) {
        $d = json_decode(file_get_contents(SETTINGS_FILE), true);
        if (is_array($d)) { $s = array_merge($s, $d); }
    }
    return $s;
}
function save_settings($s) {
    ensure_data_dir();
    file_put_contents(SETTINGS_FILE, json_encode($s, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES), LOCK_EX);
}

function st($k) { global $ST; return trim((string)($ST[$k] ?? '')); }

/* Actieve diensten: sleutel => gegevens */
function services() {
    global $ST;
    $out = [];
    foreach (($ST['diensten'] ?? []) as $d) {
        if (trim($d['naam'] ?? '') === '' || empty($d['key']) || empty($d['slug'])) continue;
        $out[$d['key']] = $d;
    }
    return $out;
}
function cat_names() {
    $o = [];
    foreach (services() as $k => $d) { $o[$k] = $d['naam']; }
    return $o;
}
function service_url($key) {
    $s = services();
    return isset($s[$key]) ? '/' . $s[$key]['slug'] . '/' : '/#diensten';
}

/* Webadres van de site: uit de instellingen, anders het huidige domein */
function site_url() {
    $u = st('site_url');
    if ($u !== '') return rtrim($u, '/');
    $host = preg_replace('/[^a-zA-Z0-9.\-:]/', '', (string)($_SERVER['HTTP_HOST'] ?? 'localhost'));
    $https = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') || (($_SERVER['HTTP_X_FORWARDED_PROTO'] ?? '') === 'https');
    return ($https ? 'https://' : 'http://') . $host;
}
function tel_link($t) {
    $d = preg_replace('/[^0-9+]/', '', (string)$t);
    if ($d === '') return '';
    if (strpos($d, '00') === 0) return '+' . substr($d, 2);
    if ($d[0] === '0') return '+31' . substr($d, 1);
    return $d;
}
function valid_color($c, $fallback) {
    return preg_match('/^#[0-9a-fA-F]{6}$/', (string)$c) ? strtolower($c) : $fallback;
}
function logo_url() {
    $l = st('logo');
    return $l !== '' ? '/uploads/site/' . rawurlencode($l) : '';
}
function chat_url() {
    $c = st('chat_url');
    return preg_match('#^https://#i', $c) ? $c : '';
}

/* Zet de gekozen stijl in een gemaakte pagina */
function stijl_pagina($html) {
    $s = stijl();
    if ($s === 'stoer') return $html;
    $pos = strpos($html, '</head>');
    if ($pos !== false) { $html = substr($html, 0, $pos) . stijl_css($s) . substr($html, $pos); }
    $pos = strpos($html, '<body>');
    if ($pos !== false) { $html = substr($html, 0, $pos) . '<body class="t-' . $s . '">' . substr($html, $pos + 6); }
    return $html;
}

function paragraphs($text) {
    $out = [];
    foreach (preg_split('/\R{2,}/', trim((string)$text)) as $p) { if (trim($p) !== '') $out[] = trim($p); }
    return $out;
}

function ensure_data_dir() {
    if (!is_dir(DATA_DIR)) { mkdir(DATA_DIR, 0755, true); }
    if (!file_exists(DATA_DIR . '/.htaccess')) { file_put_contents(DATA_DIR . '/.htaccess', "Require all denied\n"); }
}
function ensure_upload_dir() {
    if (!is_dir(UPLOAD_DIR)) { mkdir(UPLOAD_DIR, 0755, true); }
    if (!file_exists(UPLOAD_DIR . '/.htaccess')) {
        file_put_contents(UPLOAD_DIR . '/.htaccess', "<FilesMatch \"\\.(php|php\\d|phtml|phar|pl|py|cgi|sh|htaccess)$\">\nRequire all denied\n</FilesMatch>\nOptions -Indexes\n");
    }
}
function read_json($f, $default) {
    if (!file_exists($f)) return $default;
    $d = json_decode(file_get_contents($f), true);
    return is_array($d) ? $d : $default;
}
function write_json($f, $data) {
    ensure_data_dir();
    file_put_contents($f, json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES), LOCK_EX);
}

function slugify($text) {
    $text = trim($text);
    $from = ['á','à','ä','â','é','è','ë','ê','í','ì','ï','î','ó','ò','ö','ô','ú','ù','ü','û','ñ','ç',
             'Á','À','Ä','Â','É','È','Ë','Ê','Í','Ì','Ï','Î','Ó','Ò','Ö','Ô','Ú','Ù','Ü','Û','Ñ','Ç'];
    $to   = ['a','a','a','a','e','e','e','e','i','i','i','i','o','o','o','o','u','u','u','u','n','c',
             'a','a','a','a','e','e','e','e','i','i','i','i','o','o','o','o','u','u','u','u','n','c'];
    $text = str_replace($from, $to, $text);
    $text = strtolower($text);
    $text = preg_replace('/[^a-z0-9]+/', '-', $text);
    $text = trim($text, '-');
    return $text === '' ? 'pagina' : $text;
}

function unique_slug($base, $pages, $ignore = null) {
    global $RESERVED;
    $svc = service_slugs();
    $slug = $base; $i = 2;
    $taken = function($s) use ($pages, $ignore, $RESERVED, $svc) {
        if (in_array($s, $RESERVED, true) || in_array($s, $svc, true)) return true;
        foreach ($pages as $p) { if ($p['slug'] === $s && $p['slug'] !== $ignore) return true; }
        if ($s !== $ignore && file_exists(ROOT . '/' . $s)) return true;
        return false;
    };
    while ($taken($slug)) { $slug = $base . '-' . $i; $i++; }
    return $slug;
}
function rrmdir($dir) {
    if (!is_dir($dir)) return;
    foreach (scandir($dir) as $item) {
        if ($item === '.' || $item === '..') continue;
        $path = $dir . '/' . $item;
        is_dir($path) ? rrmdir($path) : @unlink($path);
    }
    @rmdir($dir);
}

/* Grote telefoonfoto's verkleinen en rechtop zetten (alleen jpg, alleen als de server dat kan) */
function shrink_photo($path) {
    $max = 2000;
    if (!function_exists('imagecreatefromjpeg') || !function_exists('exif_read_data')) return;
    $info = @getimagesize($path);
    if (!$info || $info[2] !== IMAGETYPE_JPEG) return;
    $w = $info[0]; $hgt = $info[1];
    $exif = @exif_read_data($path);
    $orient = is_array($exif) && isset($exif['Orientation']) ? (int)$exif['Orientation'] : 1;
    if (max($w, $hgt) <= $max && $orient === 1) return;
    $src = @imagecreatefromjpeg($path);
    if (!$src) return;
    if ($orient === 3) { $src = imagerotate($src, 180, 0); }
    elseif ($orient === 6) { $src = imagerotate($src, -90, 0); }
    elseif ($orient === 8) { $src = imagerotate($src, 90, 0); }
    $w = imagesx($src); $hgt = imagesy($src);
    $scale = min(1, $max / max($w, $hgt));
    $nw = (int)round($w * $scale); $nh = (int)round($hgt * $scale);
    $dst = imagecreatetruecolor($nw, $nh);
    imagecopyresampled($dst, $src, 0, 0, 0, 0, $nw, $nh, $w, $hgt);
    @imagejpeg($dst, $path, 84);
    imagedestroy($src); imagedestroy($dst);
}

/* Een geuploade afbeelding controleren en opslaan. Geeft bestandsnaam of '' terug. */
function save_one_image($tmp, $name, $dir, $prefix, &$errors) {
    $types = [IMAGETYPE_JPEG => 'jpg', IMAGETYPE_PNG => 'png', IMAGETYPE_WEBP => 'webp', IMAGETYPE_GIF => 'gif'];
    if (!is_uploaded_file($tmp)) return '';
    if (filesize($tmp) > MAX_PHOTO_BYTES) { $errors[] = "'" . $name . "' is groter dan 8 MB."; return ''; }
    $info = @getimagesize($tmp);
    if (!$info || !isset($types[$info[2]])) { $errors[] = "'" . $name . "' is geen jpg, png, webp of gif."; return ''; }
    if (!is_dir($dir)) { mkdir($dir, 0755, true); }
    $file = $prefix . date('Ymd-His') . '-' . bin2hex(random_bytes(4)) . '.' . $types[$info[2]];
    if (move_uploaded_file($tmp, $dir . '/' . $file)) { shrink_photo($dir . '/' . $file); return $file; }
    return '';
}
function upload_error_text($err, $name) {
    if ($err === UPLOAD_ERR_INI_SIZE || $err === UPLOAD_ERR_FORM_SIZE) return "'" . $name . "' is te groot voor de server.";
    return "'" . $name . "' kon niet worden geupload (foutcode " . $err . ").";
}
function save_uploaded_photos($slug, &$errors) {
    $saved = [];
    if (empty($_FILES['photos']) || !is_array($_FILES['photos']['name'])) return $saved;
    ensure_upload_dir();
    $count = count($_FILES['photos']['name']);
    for ($i = 0; $i < $count; $i++) {
        $err = $_FILES['photos']['error'][$i];
        if ($err === UPLOAD_ERR_NO_FILE) continue;
        if ($err !== UPLOAD_ERR_OK) { $errors[] = upload_error_text($err, $_FILES['photos']['name'][$i]); continue; }
        $f = save_one_image($_FILES['photos']['tmp_name'][$i], $_FILES['photos']['name'][$i], UPLOAD_DIR . '/' . $slug, '', $errors);
        if ($f !== '') $saved[] = $f;
    }
    return $saved;
}

function business_jsonld() {
    $data = [
        '@context' => 'https://schema.org',
        '@type'    => 'LocalBusiness',
        '@id'      => site_url() . '/#bedrijf',
        'name'     => st('naam'),
        'url'      => site_url() . '/',
    ];
    if (logo_url() !== '') { $data['logo'] = site_url() . logo_url(); $data['image'] = site_url() . logo_url(); }
    if (st('tel') !== '') $data['telephone'] = tel_link(st('tel'));
    if (st('email') !== '') $data['email'] = st('email');
    if (st('straat') !== '' || st('plaats') !== '') {
        $data['address'] = array_filter([
            '@type'           => 'PostalAddress',
            'streetAddress'   => st('straat'),
            'postalCode'      => st('postcode'),
            'addressLocality' => st('plaats'),
            'addressRegion'   => st('provincie'),
            'addressCountry'  => 'NL',
        ], function($v) { return $v !== ''; });
    }
    if (st('werkgebied') !== '') $data['areaServed'] = st('werkgebied');
    $names = array_values(cat_names());
    if (!empty($names)) $data['knowsAbout'] = $names;
    return '<script type="application/ld+json">' . json_encode($data, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES | JSON_HEX_TAG) . '</script>';
}

function meta_desc_for($p) {
    $d = trim($p['desc'] ?? '');
    if ($d === '') { $d = trim(preg_replace('/\s+/', ' ', $p['text'] ?? '')); }
    $names = cat_names();
    $cats = array_map(function($c) use ($names) { return $names[$c]; }, page_cats($p));
    $base = page_title($p) . (empty($cats) ? '' : ' (' . implode(', ', $cats) . ')') . '. Een project van ' . st('naam') . (st('plaats') !== '' ? ' uit ' . st('plaats') : '') . '.';
    $d = $d === '' ? $base : $d . ' - ' . $base;
    return cut($d, 155);
}

function project_item($p) {
    $photos = $p['photos'] ?? [];
    return [
        'title' => page_title($p),
        'url'   => '/' . $p['slug'] . '/',
        'desc'  => trim($p['desc'] ?? ''),
        'photo' => !empty($photos) ? photo_url($p['slug'], $photos[0]) : '',
        'date'  => $p['updated'] ?? ($p['created'] ?? ''),
        'cats'  => page_cats($p),
    ];
}
function site_lists($pages) {
    $featured = []; $projects = [];
    foreach ($pages as $p) {
        $folder = page_folder($p);
        if ($folder === '') continue;
        $item = project_item($p);
        $projects[] = $item;
        if ($folder === 'featured') { $featured[] = $item; }
    }
    $sort = function($a, $b) { return strcmp($b['date'], $a['date']); };
    usort($projects, $sort); usort($featured, $sort);
    return [$featured, $projects];
}

/* Schrijft een pagina als index.php met bovenaan de licentiepoort.
   Staat er geen licentie.php (oude installatie), dan doet de poort niets. */
function schrijf_pagina($dir, $html) {
    $pad = (rtrim($dir, '/') === ROOT) ? '__DIR__' : 'dirname(__DIR__)';
    $kop = '<?php $ffp = ' . $pad . " . '/licentie.php'; if (is_file(\$ffp)) { require \$ffp; } ?>";
    file_put_contents($dir . '/index.php', $kop . $html, LOCK_EX);
    @unlink($dir . '/index.html');
}

function write_page_files($p) {
    $dir = ROOT . '/' . $p['slug'];
    if (!is_dir($dir)) { mkdir($dir, 0755, true); }
    if (trim($p['code'] ?? '') !== '') {
        file_put_contents($dir . '/app.html', $p['code']);
        schrijf_pagina($dir, stijl_pagina(render_code_page($p)));
    } else {
        @unlink($dir . '/app.html');
        schrijf_pagina($dir, stijl_pagina(render_project_page($p)));
    }
}

/* ===========================================================
   Alles opnieuw opbouwen na een wijziging
   =========================================================== */
function write_service_pages($pages) {
    global $RESERVED;
    $projectSlugs = array_map(function($p) { return $p['slug']; }, $pages);
    $now = [];
    foreach (services() as $k => $s) {
        if (in_array($s['slug'], $projectSlugs, true) || in_array($s['slug'], $RESERVED, true)) continue;
        $dir = ROOT . '/' . $s['slug'];
        if (!is_dir($dir)) { mkdir($dir, 0755, true); }
        schrijf_pagina($dir, stijl_pagina(render_service_page($k, $pages)));
        $now[] = $s['slug'];
    }
    // Oude dienstpagina's opruimen (alleen mappen die de admin zelf maakte en waar alleen index.html in staat)
    foreach (generated_slugs() as $old) {
        if (in_array($old, $now, true) || in_array($old, $projectSlugs, true) || in_array($old, $RESERVED, true)) continue;
        if (!preg_match('/^[a-z0-9-]+$/', $old)) continue;
        $dir = ROOT . '/' . $old;
        $inhoud = is_dir($dir) ? array_values(array_diff(scandir($dir), ['.', '..'])) : [];
        if ($inhoud === ['index.html'] || $inhoud === ['index.php']) { rrmdir($dir); }
    }
    write_json(GENERATED_FILE, $now);
}
function write_contact_pages() {
    foreach (['contact' => render_contact_page(), 'bedankt' => render_thanks_page()] as $slug => $html) {
        $dir = ROOT . '/' . $slug;
        if (!is_dir($dir)) { mkdir($dir, 0755, true); }
        schrijf_pagina($dir, stijl_pagina($html));
    }
}
function write_seo_files($pages) {
    $u = site_url();
    $today = date('Y-m-d');
    $urls = [['loc' => $u . '/', 'lastmod' => $today]];
    foreach (services() as $s) { $urls[] = ['loc' => $u . '/' . $s['slug'] . '/', 'lastmod' => $today]; }
    $urls[] = ['loc' => $u . '/contact/', 'lastmod' => $today];
    foreach ($pages as $p) {
        if (page_folder($p) === '') continue;
        $d = substr((string)($p['updated'] ?? ($p['created'] ?? '')), 0, 10);
        $urls[] = ['loc' => $u . '/' . rawurlencode($p['slug']) . '/', 'lastmod' => preg_match('/^\d{4}-\d{2}-\d{2}$/', $d) ? $d : $today];
    }
    $xml = '<?xml version="1.0" encoding="UTF-8"?>' . "\n" . '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">' . "\n";
    foreach ($urls as $x) { $xml .= '  <url><loc>' . h($x['loc']) . '</loc><lastmod>' . $x['lastmod'] . '</lastmod></url>' . "\n"; }
    $xml .= '</urlset>' . "\n";
    file_put_contents(ROOT . '/sitemap.xml', $xml, LOCK_EX);
    $robots = "User-agent: *\nAllow: /\nDisallow: /admin.php\nDisallow: /data/\n\nSitemap: " . $u . "/sitemap.xml\n";
    file_put_contents(ROOT . '/robots.txt', $robots, LOCK_EX);
}
function rebuild_site($pages) {
    list($featured, $projects) = site_lists($pages);
    file_put_contents(SITE_FILE, json_encode(['featured' => $featured, 'projects' => $projects, 'generated' => date('c')], JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES), LOCK_EX);
    schrijf_pagina(ROOT, stijl_pagina(render_home($pages)));
    write_service_pages($pages);
    write_contact_pages();
    write_seo_files($pages);
}
function rebuild_everything($pages) {
    foreach ($pages as $p) { write_page_files($p); }
    rebuild_site($pages);
}

/* ===========================================================
   Licentiepoort (licentie.php)
   De admin schrijft dit bestand zelf. Elke pagina laadt het.
   Het vraagt hooguit eens per 6 uur aan de FormForge-server of
   het abonnement nog loopt. Is de server niet bereikbaar, dan
   blijft de laatst bekende status gelden (bij twijfel: open).
   =========================================================== */
define('POORT_VERSIE', '1');
function poort_code() {
    return <<<'PHP'
<?php
/* FormForge licentiepoort, versie 1. Niet aanpassen: de admin schrijft dit bestand opnieuw. */
if (!function_exists('ff_licentie_status')) {
    function ff_http_get($url, $timeout) {
        if (function_exists('curl_init')) {
            $ch = curl_init($url);
            curl_setopt_array($ch, [CURLOPT_RETURNTRANSFER => true, CURLOPT_TIMEOUT => $timeout, CURLOPT_CONNECTTIMEOUT => $timeout, CURLOPT_FOLLOWLOCATION => false]);
            $b = curl_exec($ch); $code = (int)curl_getinfo($ch, CURLINFO_HTTP_CODE); curl_close($ch);
            return ($b !== false && $code >= 200 && $code < 300) ? $b : null;
        }
        $ctx = stream_context_create(['http' => ['timeout' => $timeout, 'ignore_errors' => false]]);
        $b = @file_get_contents($url, false, $ctx);
        return $b === false ? null : $b;
    }
    function ff_licentie_status($vers = false) {
        $dir = __DIR__ . '/data';
        $lic = @json_decode((string)@file_get_contents($dir . '/licentie.json'), true);
        if (!is_array($lic) || empty($lic['key']) || !preg_match('/^[a-f0-9]{48}$/', $lic['key'])) return ['actief' => true, 'geen' => true];
        $cf = $dir . '/licentie-status.json';
        $c = @json_decode((string)@file_get_contents($cf), true);
        if (!is_array($c)) $c = [];
        $nu = time();
        if (!$vers && isset($c['actief']) && $nu - (int)($c['checked'] ?? 0) < 6 * 3600) return $c;
        if (!$vers && isset($c['actief']) && $nu - (int)($c['tried'] ?? 0) < 900) return $c;
        $server = (isset($lic['server']) && preg_match('#^https://[a-z0-9.\-]+$#i', $lic['server'])) ? $lic['server'] : 'https://formforge-sync-1.onrender.com';
        $body = ff_http_get($server . '/api/website/licentie?key=' . rawurlencode($lic['key']) . ($vers ? '&vers=1' : ''), 6);
        $d = $body !== null ? json_decode($body, true) : null;
        if (is_array($d) && isset($d['actief'])) {
            $c = ['actief' => (bool)$d['actief'], 'waarschuwing' => !empty($d['waarschuwing']), 'offlineOp' => (string)($d['offlineOp'] ?? ''), 'reden' => (string)($d['reden'] ?? ''), 'checked' => $nu];
        } elseif (!isset($c['actief'])) {
            $c['actief'] = true;
        }
        $c['tried'] = $nu;
        @file_put_contents($cf, json_encode($c), LOCK_EX);
        return $c;
    }
}
if (!defined('FF_GEEN_POORT')) {
    $ffst = ff_licentie_status(false);
    if (empty($ffst['actief'])) {
        http_response_code(503);
        header('Retry-After: 86400');
        header('Content-Type: text/html; charset=UTF-8');
        echo '<!DOCTYPE html><html lang="nl"><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta name="robots" content="noindex"><title>Tijdelijk niet beschikbaar</title></head>'
           . '<body style="margin:0;min-height:100vh;display:grid;place-items:center;background:#111827;color:#fff;font-family:Segoe UI,Arial,sans-serif;text-align:center;padding:24px">'
           . '<div><h1 style="font-size:32px;margin:0 0 10px">Deze website is tijdelijk niet beschikbaar.</h1><p style="margin:0;color:rgba(255,255,255,.75);font-size:18px">Probeer het later nog eens.</p></div></body></html>';
        exit;
    }
}
PHP;
}
function schrijf_poort() {
    $code = poort_code();
    $f = ROOT . '/licentie.php';
    if (!is_file($f) || md5_file($f) !== md5($code)) { @file_put_contents($f, $code, LOCK_EX); }
}
function licentie_key() {
    $l = read_json(DATA_DIR . '/licentie.json', []);
    return (isset($l['key']) && preg_match('/^[a-f0-9]{48}$/', $l['key'])) ? $l['key'] : '';
}
function licentie_server() {
    $l = read_json(DATA_DIR . '/licentie.json', []);
    return (isset($l['server']) && preg_match('#^https://[a-z0-9.\-]+$#i', $l['server'])) ? $l['server'] : 'https://formforge-sync-1.onrender.com';
}
function betaal_link() {
    return licentie_server() . '/api/website/betalen?key=' . rawurlencode(licentie_key());
}
function nl_datum($iso) {
    $t = strtotime((string)$iso);
    if (!$t) return '';
    $m = ['januari','februari','maart','april','mei','juni','juli','augustus','september','oktober','november','december'];
    return date('j', $t) . ' ' . $m[(int)date('n', $t) - 1] . ' ' . date('Y', $t);
}

define('RESET_FILE', DATA_DIR . '/reset.json');
function reset_geldig($token) {
    if (!preg_match('/^[a-f0-9]{40}$/', (string)$token)) return false;
    $r = read_json(RESET_FILE, []);
    return !empty($r['hash']) && (int)($r['exp'] ?? 0) > time() && hash_equals($r['hash'], hash('sha256', $token));
}
function stuur_resetmail($naar, $link) {
    // Eerst via de FormForge-server (komt betrouwbaarder aan), anders via de eigen server.
    $key = licentie_key();
    if ($key !== '' && function_exists('curl_init')) {
        $ch = curl_init(licentie_server() . '/api/website/reset-mail');
        curl_setopt_array($ch, [CURLOPT_RETURNTRANSFER => true, CURLOPT_TIMEOUT => 12, CURLOPT_POST => true,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
            CURLOPT_POSTFIELDS => json_encode(['key' => $key, 'email' => $naar, 'link' => $link])]);
        curl_exec($ch); $code = (int)curl_getinfo($ch, CURLINFO_HTTP_CODE); curl_close($ch);
        if ($code >= 200 && $code < 300) return true;
    }
    $host = preg_replace('/:\d+$/', '', preg_replace('/^www\./', '', strtolower((string)($_SERVER['HTTP_HOST'] ?? ''))));
    $van = preg_match('/^[a-z0-9.-]+\.[a-z]{2,}$/', $host) ? 'noreply@' . $host : $naar;
    $enc = function($x) { return '=?UTF-8?B?' . base64_encode($x) . '?='; };
    $tekst = "Je hebt gevraagd om een nieuw wachtwoord voor de beheerpagina van " . st('naam') . ".\r\n\r\nKlik op deze link om een nieuw wachtwoord te kiezen:\r\n" . $link . "\r\n\r\nDe link werkt 1 uur. Heb je dit niet zelf gevraagd? Dan kun je deze mail negeren.\r\n";
    $headers = 'From: ' . $enc(st('naam')) . ' <' . $van . ">\r\nMIME-Version: 1.0\r\nContent-Type: text/plain; charset=UTF-8\r\nContent-Transfer-Encoding: 8bit\r\n";
    return @mail($naar, $enc('Nieuw wachtwoord voor je website'), $tekst, $headers, '-f' . $van);
}

function login_key() { return hash('sha256', ($_SERVER['REMOTE_ADDR'] ?? '') . 'ff-login'); }
function login_blocked() {
    $l = read_json(LOGIN_LIMIT_FILE, []);
    $now = time();
    $recent = array_filter((array)($l[login_key()] ?? []), function($t) use ($now) { return $now - (int)$t < 900; });
    return count($recent) >= 10;
}
function login_failed() {
    $l = read_json(LOGIN_LIMIT_FILE, []);
    $now = time();
    foreach ($l as $k => $v) {
        $l[$k] = array_values(array_filter((array)$v, function($t) use ($now) { return $now - (int)$t < 900; }));
        if (empty($l[$k])) unset($l[$k]);
    }
    $l[login_key()][] = $now;
    write_json(LOGIN_LIMIT_FILE, $l);
}
